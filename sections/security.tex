\section{Security}

We next prove the security of $\pedpopplus$ with respect to the notion of security defined by $\FDKGAbort$.

\begin{theorem}
\label{thm:keygen-pedpopplus}
Assume $\NIZKUCExtractable$ is a secure online-extractable (straight-line extractable) zero-knowledge
proof for the relation $\mathcal{R}_P$ of Section~\ref{def:online-extractable}.
Then, provided the adversary corrupts at most $\lvert\corrupt\rvert \leq \tcor$ parties
and the network satisfies the fault bound of the reliable broadcast channel
(Section~\ref{sec:comm-model}),
the $\pedpopplus$ DKG is secure under aborts in the programmable random oracle model,
and realizes the idealized notion $\FDKGAbort$.
\end{theorem}

\begin{proof}
    To establish the proof of Theorem~\ref{thm:keygen-pedpopplus},
    we define a simulator $\SIM$ that is given black-box access to the static adversary $\adv$.

%     aggregate distinguishing advantage, reducing to the ZK/online-extractability of $\mathcal{R}_P$ and the
%     relevant hardness assumption. The single ``statistically indistinguishable within $q_h/q$'' claim below

    \medskip

    \paragraph{\bf Setup.}
    \begin{itemize}
        \item $\SIM$ runs the adversary,
        and receives the set $\corrupt$.
        $\SIM$ then sets $\honest \gets \set{n} \setminus \corrupt$.
        \item $\SIM$ outputs $\corrupt$ to the functionality $\FDKGAbort$.
        \item $\SIM$ receives the DKG challenge $\dkgchal \in \Gr$ from $\FDKGAbort$.
        \item $\SIM$ sets up the verifiable-encryption trapdoor: it samples $\eta \rgets \Zq$, sets
        $h \gets g^{\eta}$, and publishes $h$ as the random group element of the proof system,
        keeping $\eta$ secret so that it can later extract by decryption (online, without rewinding).
        \item $\SIM$ initializes the table $\QSchnorr \gets \emptyset$
    \end{itemize}

    \medskip

    \paragraph{\bf Simulating Random Oracle Queries.}
    When queried by the adversary,
    $\SIM$ simulates its random oracles by lazy sampling.
    \begin{itemize}
      \item $\HashSid$
      \item $\HashCom$
      \item $\HashSchnorrR$
    \end{itemize}

    \medskip

    \paragraph{\bf Round 0.}
    To simulate the initial round of the DKG,
    $\SIM$ follows the protocol honestly for each party $i \in \honest$.

    \medskip

    \paragraph{\bf Round 1.}
    On input $\sid_1, \ldots, \sid_n$,
    to simulate Round~1 of the DKG,
    $\SIM$ does the following:
    \begin{itemize}
    \item Derive $\sid$ honestly by querying the random oracle $\HashSid$ on $\sid_1, \ldots, \sid_n$.
    \item  For each honest party $i \in \honest$,
    sample $h_i \rgets \Zq$.
    \item When the adversary queries party $k \in \honest$,
    output $h_k$.
    \end{itemize}

    \medskip

    \paragraph{\bf Round 2.}
    On input $(\{h_i\}_{i\in\set{n}})$,
    to simulate Round~2 of the DKG,
    $\SIM$ does the following:
    \begin{itemize}
      \item For each corrupt party $j \in \corrupt$,
      look up a previously input tuple $(\sid, j, C_j)$ for the output $h_j$ using the random oracle $\HashCom$.
      \item If any $h_j$ does not correspond to a valid entry or to the current $\sid$,
      follow the remainder of Round~2 honestly for all honest participants.
      There is no need to simulate as honest parties will abort in a later
      round due to invalid validity checks.
      \item Otherwise, pick at random an identifier $\simid \rgets \honest$,
      and follow Round~2 as defined in the protocol
      for all honest parties $i \in \honest, i \neq \simid$,
      \item However, for party $\simid$,
        simulate Round~2,
        as follows:
        \begin{enumerate}
    \item Set
          \begin{equation}
            \label{eq:simid-comm}
C_{\simid}[0] \gets \dkgchal \cdot
              \prod_{k \in \honest, k  \neq \simid} (C_{k}[0])^{-1} \cdot
              \prod_{j \in \corrupt} (C_{j}[0])^{-1}
          \end{equation}
          where $C_{k}[0]$ are from the honest parties $k \in \honest, k \neq \simid$.
        \item Randomly sample corrupt parties' shares $\share_{\simid,j} \rgets \Zq$ for all $j \in \corrupt$.
        \item For $\ell \in \set{\tcor}$, set the remaining commitments, as follows:
          \[ C_\simid[\ell] \gets C_\simid[0]^{\lambda'_{0,\ell}} \cdot
            \prod_{j \in \corrupt } (g^{\share_{\simid,j}})^{\lambda'_{j,\ell}} \]
        where $\lambda'_{0,\ell}$ and $\lambda'_{j,\ell}$ denote the coefficient of $x^\ell$ in the Lagrange basis
        polynomials $L'_0$ and $L'_j$ for the node set $\{ 0\}\cup\corrupt$ (See Section~\ref{prelim:secret-sharing}).
        This makes $C_\simid[\ell] = g^{a_\ell}$ a commitment to the $\ell$-th \emph{coefficient} $a_\ell$ of $f_\simid$,
        as required by the protocol's commitment vectors and the Round~4 check $g^{f_i(k)} = \prod_{\ell} C_i[\ell]^{k^\ell}$.
%         \simon{TODO: the node set $\{0\}\cup\corrupt$ determines a degree-$\tcor$ polynomial uniquely only when
%         $\lvert\{0\}\cup\corrupt\rvert = t$, i.e.\ $\lvert\corrupt\rvert = \tcor$. State this
%         worst-case-corruption assumption, and for $\lvert\corrupt\rvert < \tcor$ specify how the interpolation
%         set is completed (e.g.\ pad with random shares on honest indices).}
        \item Produce a simulated proof $\pi_\simid$ for the statement $\PK_\simid \coloneqq C_\simid[0]$
          with context $\gamma_\simid \gets (\sid, \simid, C_\simid[0])$.
          Since $C_\simid[0]$ is defined in Equation~\eqref{eq:simid-comm} so as to embed the challenge
          $\dkgchal$, $\SIM$ does not know the witness $\sk$; it instead runs the honest-verifier
          zero-knowledge simulator of $\mathcal{R}_P$ (Figure~\ref{fg:concrete-simulator}),
          reprogramming the random oracle $\HashSchnorrR$:
          \begin{enumerate}
            \item Sample $r \rgets \Zq$ and form a genuine encryption of $g^{\sk} = \PK_\simid$:
              set $C_0 \gets g^{r}$ and $C_1 \gets h^{r}\cdot \PK_\simid$. ($\SIM$ knows the group
              element $\PK_\simid$, though not the scalar $\sk$.)
            \item Sample a challenge $c' \rgets \Zq$ and responses $z_1, z_2 \rgets \Zq$ uniformly and
              independently at random.
            \item Compute the commitments forced by the verification equations:
              \[
                T_1 \gets g^{z_1}\cdot \PK_\simid^{-c'}, \qquad
                T_2 \gets g^{z_2}\cdot C_0^{-c'}, \qquad
                T_3 \gets h^{z_2}\cdot g^{z_1}\cdot C_1^{-c'}.
              \]
            \item Reprogram the random oracle so that
              $\HashSchnorrR(g, h, \PK_\simid, C_0, C_1, T_1, T_2, T_3, \gamma_\simid) \gets c'$,
              and set $\pi_\simid \gets (C_0, C_1, T_1, T_2, T_3, z_1, z_2)$.
          \end{enumerate}
        \item When the adversary queries party $k \in \honest$, output $(C_k, \pi_k)$.
      \end{enumerate}
    \end{itemize}

    \medskip

    \paragraph{\bf Bad event: reprogramming failure.}
    The last step above reprograms $\HashSchnorrR$ at the point
    $(g, h, \PK_\simid, C_0, C_1, T_1, T_2, T_3, \gamma_\simid)$.
    This is sound only if $\adv$ has not already queried $\HashSchnorrR$ at that point; otherwise $\SIM$
    would be forced to assign two distinct outputs to the same input, an inconsistency detectable by
    $\adv$, and $\SIM$ aborts. Since $C_0 = g^{r}$ with $r \rgets \Zq$ fresh and independent of the
    adversary's view prior to the proof, it is uniformly distributed in $\Gr$, so the reprogramming
    point is unpredictable to $\adv$. Letting $q_h$ bound the number of
    queries $\adv$ makes to $\HashSchnorrR$, the probability of this collision is at most
    $q_h / \lvert \Gr \rvert = q_h / q$, which is negligible in $\lambda$. Conditioned on the collision
    not occurring, the simulated proof $\pi_\simid$ is distributed identically to one produced by an
    honest prover, so the adversary's view in this hybrid is statistically indistinguishable
    (within $q_h/q$) from its view in the real protocol.

    \paragraph{\bf Round 3.}
    On input $\KeyGen_3(\{C_i, \pi_i\}_{i\in\set{n}})$,
    $\SIM$ does the following:
    \begin{itemize}
      \item Perform verification steps honestly;
        if any verification step fails, abort the protocol.
      \item When the adversary queries party $k \in \honest\setminus\{\simid\}$, output $f_k(j)$ extracted from the honest participant.
        \item Otherwise, for each corrupt party $j \in \corrupt$, extract its committed contribution by
          decrypting the verifiable-encryption ciphertext $c_j = (C_{j,0}, C_{j,1})$ contained in $\pi_j$
          using the trapdoor $\eta$ (where $h = g^{\eta}$): compute $g^{f_j(0)} \gets C_{j,1} \cdot C_{j,0}^{-\eta}$,
          from which $f_j(0)$ is determined. This extraction is straight-line (no rewinding), as required in the UC setting.
%           \simon{TODO: decryption recovers the group element $g^{f_j(0)}$, not the scalar $f_j(0)$ (recovering
%           the scalar is discrete log). Clarify what $\SIM$ actually needs: if only group elements, rewrite the
%           downstream use --- including the bias $p(\cdot)$ sent to $\FDKGAbort$, which is a scalar polynomial
%           --- accordingly; otherwise use a verifiable encryption whose extractable plaintext is the scalar
%           witness.}
        \item For party $\simid$, set $f_\simid(j) \gets \share_{\simid,j}$
        (By definition $j \in \corrupt$ is an interpolation node of $f_\simid$;
        we have $f_\simid(j) = \share_{\simid,j}$).
    \end{itemize}

    \medskip

    \paragraph{\bf Round 4.}
    On input $(\{f_i(k)\}_{i\in\set{n}})$,
    $\SIM$ does the following:
    \begin{itemize}
      \item Perform the checks as required by Round 4 for every $i\in\honest$.
      \item Skip computing the key pair.
      \item If any check fails then abort the protocol.
      \item Otherwise, for every $i\in\honest$, broadcast $(\success, \sid)$ on that party's
        terminating reliable broadcast instance $\TRB$. (A faulty or timed-out sender's
        instance delivers $\bot$ to every party.)
    \end{itemize}

    \paragraph{\bf Round 5.}
    To simulate Round~5, $\SIM$ obtains the $n$ delivered values $\{b_i\}_{i\in\set{n}}$ of the
    terminating reliable broadcast instances. If $b_i = (\success, \sid)$ for every $i\in\set{n}$, then $\SIM$
    instructs $\FDKGAbort$ to deliver the output keys; otherwise (some $b_i = \bot$) $\SIM$ instructs
    $\FDKGAbort$ to abort.

%     is the central real/ideal link and is currently absent (see also the scalar-extraction note in Round~3).}

\end{proof}
